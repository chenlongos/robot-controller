## 一、核心类型定义

在定义机器人类之前，需要先定义核心的数据类型。

```python
# src/types.py
from typing import TypedDict, List, Dict, Optional
import numpy as np

class Observation(TypedDict):
    """机器人观测数据格式
    
    用于描述机器人在某一时刻的状态信息，包含传感器数据和运动状态。
    """
    camera: np.ndarray              # 摄像头图像
    joint_positions: List[float]    # 关节位置
    gripper_position: float         # 夹爪位置 (0-1, 0为闭合)
    base_velocity: Dict[str, float] # 底盘速度 {"x": x_vel, "y": y_vel, "theta": theta_vel}
    timestamp: float                # 时间戳 (秒)

class BaseAction(TypedDict):
    """底盘动作指令格式"""
    x: float    # 前后方向速度 (m/s)
    y: float    # 左右方向速度 (m/s)
    w: float    # 旋转角速度 (rad/s)

class ArmAction(TypedDict):
    """机械臂动作指令格式"""
    joints: List[float]  # 关节目标位置
    gripper: float       # 夹爪目标位置 (0-1)

class Action(TypedDict):
    """机器人动作指令格式
    
    用于描述机器人的运动指令，包含底盘和机械臂两部分。
    """
    base: Optional[BaseAction]  # 底盘动作（可选）
    arm: Optional[ArmAction]    # 机械臂动作（可选）

class RobotState(Enum):
    """机器人状态枚举"""
    INIT = "init"              # 初始化状态
    SEARCH_TENNIS = "search_tennis"    # 搜索网球
    APPROACH_TENNIS = "approach_tennis" # 接近网球（远距离）
    TRACK_TENNIS = "track_tennis"      # 跟踪网球（精细调整）
    PICK = "pick"              # 抓取网球
    SEARCH_BUCKET = "search_bucket"    # 寻找桶
    APPROACH_BUCKET = "approach_bucket" # 接近桶（远距离）
    TRACK_BUCKET = "track_bucket"      # 跟踪桶（精细调整）
    PUT_BALL = "put_ball"      # 放球
    ERROR = "error"            # 错误状态
```

class Robot:
    """机器人类"""
    
    def __init__(self, config) -> None:
        self.config = config
        self.current_state: RobotState = RobotState.INIT
        self.previous_state: RobotState = RobotState.INIT
        self.error_message: Optional[str] = None
    
    def connect(self) -> None:
        """建立硬件连接"""
        pass
    
    def disconnect(self) -> None:
        """断开硬件连接"""
        pass
    
    def get_observation(self) -> Observation:
        """获取机器人状态观测"""
        pass
    
    def send_action(self, action: Action) -> None:
        """发送动作指令"""
        pass
    
    def get_state(self) -> RobotState:
        """获取当前状态"""
        return self.current_state
    
    def set_state(self, state: RobotState) -> None:
        """设置当前状态"""
        self.previous_state = self.current_state
        self.current_state = state
    
    def set_error(self, message: str) -> None:
        """设置错误状态"""
        self.error_message = message
        self.current_state = RobotState.ERROR
    
    def reset(self) -> None:
        """重置机器人"""
        self.current_state = RobotState.INIT
        self.previous_state = RobotState.INIT
        self.error_message = None
    
    def stop(self) -> None:
        """停止所有运动"""
        pass

| 当前状态 | 下一状态 | 转换条件 |

| TRACK_BUCKET | PUT_BALL | 到达放球位置 |

            observation = robot.get_observation()
            perception_time = time.perf_counter() - perception_start
            
            # ========== 决策阶段 ==========
            decision_start = time.perf_counter()
            current_state = robot.get_state()
            next_state = state_transition(current_state, observation)
            action = generate_action(current_state, observation, config)
            decision_time = time.perf_counter() - decision_start
            
            # ========== 频率控制 ==========
            # 计算感知+决策的总时间
            processing_time = perception_time + decision_time
            
            # 根据处理时间动态调整执行速度
            # 如果处理时间过长，需要降低底盘速度以保证控制精度
            speed_scale = calculate_speed_scale(processing_time, NOMINAL_DELTA_T, MAX_DELTA_T)
            
            # 应用速度缩放
            if 'base' in action:
                action['base']['x'] *= speed_scale
                action['base']['y'] *= speed_scale
                action['base']['w'] *= speed_scale
            
            # ========== 执行阶段 ==========
            execution_start = time.perf_counter()
            robot.send_action(action)
            execution_time = time.perf_counter() - execution_start
            
            # 更新状态
            robot.set_state(next_state)
            
            # 计算循环时间并控制帧率
            cycle_time = time.perf_counter() - cycle_start_time
            sleep_time = max(NOMINAL_DELTA_T - cycle_time, 0.0)
            time.sleep(sleep_time)
            
            # 记录性能数据（可选）
            log_performance(cycle_time, perception_time, decision_time, execution_time, speed_scale)
            
    finally:
        robot.disconnect()

def calculate_speed_scale(processing_time: float, nominal_delta_t: float, max_delta_t: float) -> float:
    """计算速度缩放因子
    
    Args:
        processing_time: 感知+决策的时间
        nominal_delta_t: 标称时间间隔
        max_delta_t: 最大允许时间间隔
    
    Returns:
        速度缩放因子 (0.0 ~ 1.0)
    """
    if processing_time <= nominal_delta_t:
        return 1.0  # 处理时间正常，全速运行
    
    if processing_time >= max_delta_t:
        return 0.0  # 处理时间过长，停止移动
    
    # 线性插值计算缩放因子
    return 1.0 - (processing_time - nominal_delta_t) / (max_delta_t - nominal_delta_t)

### 3.3 频率控制原理

频率控制的核心思想是：**如果感知和决策的时间会影响执行的效果，则执行时要依据感知和决策的时间，控制底盘的运行速度。**

**控制策略：**

| 处理时间 | 速度缩放因子 | 说明 |
|----------|-------------|------|
| ≤ 标称间隔 | 1.0 | 处理时间正常，全速运行 |
| 标称间隔 ~ 最大间隔 | 1.0 ~ 0.0 | 线性降低速度 |
| ≥ 最大间隔 | 0.0 | 处理超时，停止移动 |

**数学公式：**

```
speed_scale = 1.0, 当 processing_time ≤ nominal_delta_t
speed_scale = 1.0 - (processing_time - nominal_delta_t) / (max_delta_t - nominal_delta_t), 当 nominal_delta_t < processing_time < max_delta_t
speed_scale = 0.0, 当 processing_time ≥ max_delta_t
```

**参数说明：**
- `TARGET_FPS`：目标帧率，默认30fps
- `NOMINAL_DELTA_T`：标称时间间隔，等于 1/TARGET_FPS
- `MAX_DELTA_T`：最大允许时间间隔，超过此值则停止移动
